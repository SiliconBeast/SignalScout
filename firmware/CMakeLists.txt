# Host build without PlatformIO: unit tests + simulator.
#   cmake -B build && cmake --build build -j && ctest --test-dir build
#   build/sim --shots ../docs/screenshots
# Add -DSS_SDL=ON for the interactive window (needs libsdl2-dev).
cmake_minimum_required(VERSION 3.16)
project(signalscout_host C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 99)
option(SS_SDL "Interactive SDL window" OFF)
include(FetchContent)
if(NOT LVGL_DIR)
  FetchContent_Declare(lvgl_src GIT_REPOSITORY https://github.com/lvgl/lvgl.git GIT_TAG v9.2.2 GIT_SHALLOW ON)
  FetchContent_Populate(lvgl_src)
  set(LVGL_DIR ${lvgl_src_SOURCE_DIR})
endif()
if(NOT UNITY_DIR)
  FetchContent_Declare(unity_src GIT_REPOSITORY https://github.com/ThrowTheSwitch/Unity.git GIT_TAG v2.6.0 GIT_SHALLOW ON)
  FetchContent_Populate(unity_src)
  set(UNITY_DIR ${unity_src_SOURCE_DIR})
endif()

file(GLOB CORE lib/ss_core/*.cpp lib/ss_services/*.cpp)
add_library(ss STATIC ${CORE})
target_include_directories(ss PUBLIC lib/ss_core lib/ss_services lib/ss_mock include)
target_compile_options(ss PRIVATE -Wall -Wextra)

add_executable(test_core test/test_core/test_main.cpp ${UNITY_DIR}/src/unity.c)
target_include_directories(test_core PRIVATE ${UNITY_DIR}/src)
target_compile_definitions(test_core PRIVATE UNITY_INCLUDE_DOUBLE)
target_link_libraries(test_core ss)
enable_testing()
add_test(NAME core COMMAND test_core WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})

file(GLOB_RECURSE LVGL_SRC ${LVGL_DIR}/src/*.c)
add_library(lvgl STATIC ${LVGL_SRC})
target_include_directories(lvgl PUBLIC ${LVGL_DIR} include)
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE)
add_executable(sim src/sim/main.cpp lib/ss_ui/ui.cpp)
target_include_directories(sim PRIVATE lib/ss_ui)
target_compile_options(sim PRIVATE -Wall -Wextra)
target_link_libraries(sim ss lvgl m)
if(SS_SDL)
  find_package(SDL2 REQUIRED)
  target_compile_definitions(lvgl PUBLIC SS_SDL)
  target_link_libraries(sim SDL2::SDL2)
endif()
